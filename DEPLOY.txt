## STEP 1 — Upload Files

rsync -avz --delete ./dnd-final/ root@144.126.155.14:/var/www/html/grimoire/


## STEP 2 — Permissions

ssh root@144.126.155.14
chmod -R 755 /var/www/html/grimoire/
chown -R www-data:www-data /var/www/html/grimoire/


## STEP 3 — Apache Virtual Host

nano /etc/apache2/sites-available/grimoire.conf

# ─────────────────────────────────────────────────────────────
<VirtualHost *:80>
    ServerName vg.highfunctioningchaos.living
    DocumentRoot /var/www/html/grimoire

    <Directory /var/www/html/grimoire>
        Options -Indexes +FollowSymLinks
        AllowOverride All
        Require all granted
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/grimoire_error.log
    CustomLog ${APACHE_LOG_DIR}/grimoire_access.log combined
</VirtualHost>
# ─────────────────────────────────────────────────────────────

a2ensite grimoire.conf
a2enmod rewrite
systemctl reload apache2


## STEP 4 — SSL

certbot --apache -d vg.highfunctioningchaos.living


## STEP 5 — Cloudflare DNS

# Type: A  |  Name: d&d-vg  |  Content: 144.126.155.14  |  Proxy: ON
